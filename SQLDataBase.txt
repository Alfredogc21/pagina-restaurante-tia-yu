CREATE TABLE roles (
    idRoles INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    roles CHAR(30)
);

CREATE TABLE estadoUsuario (
    idEstadoUsuario INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    estado VARCHAR(12) NOT NULL
);

CREATE TABLE disponibilidadMesa (
    idDisponibilidadMesa INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    estado VARCHAR(12) NOT NULL
);

CREATE TABLE accesos (
    idAcceso INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    tipoAcceso INT(11) NOT NULL,
    descripcion VARCHAR(255)
);

CREATE TABLE mesa (
    idMesa INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    disponibilidad INT(11) NOT NULL,
    FOREIGN KEY (disponibilidad) REFERENCES disponibilidadMesa(idDisponibilidadMesa)
);

CREATE TABLE usuarios (
    idUsuarios INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    cedula VARCHAR(20),
    nombres VARCHAR(60) NOT NULL,
    apellido VARCHAR(60) NOT NULL,
    idRoles INT(11), 
    idAccesos INT(11), 
    idDisponMesa INT(11),
    idEstadoUsuario INT(11),
    telefono VARCHAR(45) NOT NULL,
    correoElectronico VARCHAR(45) NOT NULL,
    password VARCHAR(255) NOT NULL,
    fechaRegistro DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    FOREIGN KEY (idRoles) REFERENCES roles(idRoles),
    FOREIGN KEY (idAccesos) REFERENCES accesos(idAcceso),
    FOREIGN KEY (idDisponMesa) REFERENCES disponibilidadMesa(idDisponibilidadMesa),
    FOREIGN KEY (idEstadoUsuario) REFERENCES estadoUsuario(idEstadoUsuario)

);

CREATE TABLE reserva (
    idReserva INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    hora TIME,
    fecha DATE,
    comentario VARCHAR(100),
    numPersonas INT(11),
    idUsuarios INT(11),
    idMesa INT(11),
    FOREIGN KEY (idUsuarios) REFERENCES usuarios(idUsuarios),
    FOREIGN KEY (idMesa) REFERENCES mesa(idMesa)
);

CREATE TABLE valoracion (
    idValoracion INT(11) PRIMARY KEY NOT NULL AUTO_INCREMENT,
    comentario LONGTEXT,
    estrellas INT(11),
    idUsuarios INT(11),
    fechaValoracion DATETIME DEFAULT CURRENT_TIMESTAMP NOT NULL,
    FOREIGN KEY (idUsuarios) REFERENCES usuarios(idUsuarios)
);
